def _calculate_price_with_quantity_discount(price, quantity, discount_rate):
    """
    根据数量和折扣率计算价格。
    如果数量大于10，则应用折扣。
    """
    if quantity > 10:
        return price * quantity * (1 - discount_rate)
    else:
        return price * quantity

def apply_discount(price, discount_rate):
    """
    对给定价格应用折扣。
    """
    return price * (1 - discount_rate)

def calculate_item_total(price, quantity, discount_rate):
    """
    计算单个商品的总价，包含数量折扣。
    """
    return _calculate_price_with_quantity_discount(price, quantity, discount_rate)

def calculate_order_total(item_price, item_quantity, customer_discount):
    """
    计算订单的总价，包含数量折扣。
    """
